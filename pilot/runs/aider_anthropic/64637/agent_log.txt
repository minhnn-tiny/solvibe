     /// @dev Anybody can cancel an expired bet and send funds back to each     
 party. The maker can cancel a pending bet.                                     
     function cancel() external {                                               
         IBet.Bet memory b = _bet;                                              
         IBet.Status currentStatus = _status(b);                                
                                                                                
         // Check if the bet can be cancelled                                   
         if (currentStatus == IBet.Status.RESOLVED || currentStatus ==          
 IBet.Status.CANCELLED) {                                                       
             revert InvalidStatus();                                            
         }                                                                      
                                                                                
         // Check authorization based on status                                 
         if (currentStatus == IBet.Status.PENDING) {                            
             // Only maker can cancel a pending bet                             
             if (msg.sender != b.maker) {                                       
                 revert Unauthorized();                                         
             }                                                                  
         } else if (currentStatus == IBet.Status.ACTIVE) {                      
             // Only judge can cancel an active bet                             
             if (msg.sender != b.judge) {                                       
                 revert Unauthorized();                                         
             }                                                                  
         }                                                                      
         // For EXPIRED status, anyone can cancel (no check needed)             
                                                                                
         // Withdraw from Aave if pool is set                                   
         uint256 makerRefund = b.makerStake;                                    
         uint256 takerRefund = b.takerStake;                                    
                                                                                
         if (address(_aavePool) != address(0)) {                                
             uint256 aTokenBalance =                                            
 IERC20(_aavePool.getReserveAToken(b.asset))                                    
                 .balanceOf(address(this));                                     
                                                                                
             if (aTokenBalance > 0) {                                           
                 _aavePool.withdraw(b.asset, aTokenBalance, address(this));     
                                                                                
                 // Cap refunds at what was actually withdrawn                  
                 uint256 totalStakes = b.makerStake + b.takerStake;             
                 makerRefund = _min(makerRefund, (aTokenBalance * b.makerStake) 
 / totalStakes);                                                                
                 takerRefund = _min(takerRefund, (aTokenBalance * b.takerStake) 
 / totalStakes);                                                                
             }                                                                  
         }                                                                      
                                                                                
         // Return stakes to each party with try/catch to prevent one failure   
 from blocking the other                                                        
         if (makerRefund > 0) {                                                 
             try IERC20(b.asset).transfer(b.maker, makerRefund) {} catch {}     
         }                                                                      
                                                                                
         if (takerRefund > 0 && currentStatus != IBet.Status.PENDING) {         
             // Only refund taker if they actually deposited (i.e., bet was     
 accepted)                                                                      
             try IERC20(b.asset).transfer(b.taker, takerRefund) {} catch {}     
         }                                                                      
                                                                                
         // Update status and emit event                                        
         _bet.status = IBet.Status.CANCELLED;                                   
         emit BetCancelled();                                                   
     }

The implementation:                                                             

 1 Gets current status using the _status helper to properly handle expired bets 
 2 Checks authorization:                                                        
    • PENDING: only maker can cancel                                            
    • ACTIVE: only judge can cancel                                             
    • EXPIRED: anyone can cancel                                                
    • RESOLVED/CANCELLED: reverts with InvalidStatus()                          
 3 Withdraws from Aave if a pool is set, proportionally capping refunds based on
   what's actually available                                                    
 4 Returns stakes using try/catch blocks so one failed transfer doesn't block   
   the other. Only refunds the taker if the bet was accepted (not PENDING)      
 5 Updates state to CANCELLED and emits BetCancelled()
