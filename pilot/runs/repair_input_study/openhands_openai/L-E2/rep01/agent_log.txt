from itertools import combinations

def compress(t, s):
    # floor division toward -inf, matching Solidity TickLib.compress
    q = t // s
    return q

def position(c):
    # signed right shift by 8 for int24-like values
    word = c >> 8
    bit = c & 0xff
    return word, bit

def to_tick(word, bit, s):
    return (word * 256 + bit) * s

def build_words(init_ticks, s):
    words = {}
    for t in init_ticks:
        c = compress(t, s)
        w,b = position(c)
        words[w] = words.get(w, 0) | (1 << b)
    return words

def next_bit_pos_lte(word, bit):
    mask = (1 << (bit+1)) - 1
    m = word & mask
    if m == 0:
        return False, 0
    return True, m.bit_length() - 1

def advance_down(current_tick, end_tick, words, s):
    while True:
        c = compress(current_tick - 1, s)
        w,b = position(c)
        word = words.get(w, 0)
        if b == 255:
            word = words.get(w, 0)
        init, nb = next_bit_pos_lte(word, b)
        current_tick = to_tick(w, nb, s)
        if init:
            return current_tick
        if not (end_tick < current_tick):
            return current_tick

def iter_down(init_ticks, s, start, end):
    if end > start:
        raise ValueError
    if start == end:
        return []
    words = build_words(init_ticks, s)
    current = start
    current = advance_down(current, end, words, s)
    out = []
    while current > end:
        out.append(current)
        current = advance_down(current, end, words, s)
    return out
